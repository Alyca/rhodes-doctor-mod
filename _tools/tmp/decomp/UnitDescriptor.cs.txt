	public class FactionSwitcher
	{
		private bool IsStarted;

		public UnitDescriptor UnitDescriptor { get; private set; }

		public BlueprintFaction Faction { get; private set; }

		public bool ResetAttackFactions { get; private set; }

		public FactionSwitcher(UnitDescriptor unitDescriptor, BlueprintFaction faction, bool resetAttackFactions)
		{
			UnitDescriptor = unitDescriptor;
			Faction = faction;
			ResetAttackFactions = resetAttackFactions;
		}

		public void Start()
		{
			if (!IsStarted)
			{
				IsStarted = true;
				UnitDescriptor.SwitchFactions(Faction, ResetAttackFactions, permanently: false);
				UnitDescriptor.FactionSwitchers.Add(this);
			}
		}

		public void Stop()
		{
			if (IsStarted)
			{
				IsStarted = false;
				UnitDescriptor.FactionSwitchers.Remove(this);
				BlueprintFaction faction = ((UnitDescriptor.FactionSwitchers.Count > 0) ? UnitDescriptor.FactionSwitchers.Last().Faction : UnitDescriptor.FactionOriginal);
				UnitDescriptor.SwitchFactions(faction, ResetAttackFactions, permanently: false);
			}
		}
	}

	public bool IsPlayerFaction
	{
		get
		{
			if (!m_IsPlayerFactionCached.HasValue)
			{
				m_IsPlayerFactionCached = Faction == BlueprintRoot.Instance.PlayerFaction;
			}
			if (!m_IsPlayerFactionCached.Value || AttackFactions.IsPlayerEnemy)
			{
				return false;
			}
			return true;
		}
	}

		get

			{
				return State.HasCondition(UnitCondition.TrueSeeing);

			return true;
		}
	}

	public bool IsSeeInvisibility
	{
		get
		{
			if (!State.HasCondition(UnitCondition.SeeInvisibility))
			{
				return State.HasCondition(UnitCondition.TrueSeeing);

			return true;
		}
	}

	public void TurnOn()
	{
		IsTurnedOn = true;
		Progression.CheckStatPrerequisitesAndUpdate();
		if (Inventory.OwnerUnit == this)
		{
			Inventory.TurnOn();
		}
		Body.TurnOn();
		Brain.TurnOn();
	}

	public void TurnOff()
	{
		IsTurnedOn = false;
		if (Inventory.OwnerUnit == this)
		{
			Inventory.TurnOff();
		}
		Body.TurnOff();
		Brain.TurnOff();
	}

	public void PreSave()
	{
		Body.PreSave();
		Progression.PreSave();
		foreach (Spellbook value in m_Spellbooks.Values)
		{
			value.PreSave();
		}
		if (Inventory.OwnerUnit == this)
		{
			Inventory.PreSave();
		}
		Brain.PreSave();
	}

	public void PostLoad(UnitEntityData entity)
	{
		Unit = entity;
		if (OriginalBlueprint == null)
		{
			OriginalBlueprint = Blueprint;
		}
		Stats.PostLoad(this);
		Alignment.PostLoad(this);
		Resources.PostLoad(this);
		if (!ContextData<UnitEntityData.Respec>.Current)
		{
			Body.PostLoad(this);
		}
		Progression.PostLoad(this);
		foreach (Spellbook value in m_Spellbooks.Values)
		{
			value.PostLoad(this);
		}
		State.PostLoad(this);
		AttackFactions.PostLoad(this);
		if (Unit == GameHelper.GetPlayerCharacter())
		{
			Brain.SetBrain(Blueprint.DefaultBrain);
			Brain.RestoreAvailableActions();
		}
		Brain.PostLoad(this);
	}

	public void OnComponentsDidPostLoad()
	{
		UISettings.PostLoad(this);
	}

	public void ApplyPostLoadFixes()
	{
		if (m_LoadedDollData != null)
		{
			Unit.Ensure<UnitPartDollData>().SetDefault(m_LoadedDollData);
		}
		OnDifficultyChanged();
		EnsureNaturalProficiencies();
		Stats.ApplyPostLoadFixes();
		Body.ApplyPostLoadFixes();
		State.ApplyPostLoadFixes();
		if (IsMainCharacter && !m_IsEssentialForGame)
		{
			m_IsEssentialForGame.Retain();
		}
		foreach (var (key, spellbook2) in m_Spellbooks.ToTempList())
		{
			if (spellbook2.RawBaseLevel < 1 && spellbook2.MythicLevel < 1)
			{
				m_Spellbooks.Remove(key);
			}
			else
			{
				spellbook2.ApplyPostLoadFixes();
			}
		}
		AddPostLoadActions component = Blueprint.GetComponent<AddPostLoadActions>();
		if (component != null && component.Actions.HasActions)
		{
			using (new MechanicsContext(Unit, this, Blueprint).GetDataScope())
			{
				component.Actions.Run();
			}
		}
		Blueprint.CallComponents(delegate(FixUnitOnPostLoad a)
		{
			if (!m_AppliedFixers.HasItem(a.name))
			{
				a.OnPostLoad(this);
				m_AppliedFixers = m_AppliedFixers ?? new List<string>();
				m_AppliedFixers.Add(a.name);
				PFLog.Default.Log($"Fixer applied to unit: {this} ({a.name})");
			}
		});
		if (!State.IsFinallyDead)
		{
			RemoveBuffsOnResurrect();
		}
		if (AttackFactions.Contains(Faction))
		{
			AttackFactions.Remove(Faction);
			PFLog.Default.Error("AttackFactions of unit can't contains unit's own Faction, fixed ({0})", this);
		}
		if (Body.IsPolymorphed && !Facts.List.HasItem((EntityFact i) => i.GetComponent<Polymorph>() != null))
		{
			Body.CancelPolymorphEffect();
		}
		Progression.ApplyPostLoadFixes();
	}

	private void EnsureNaturalProficiencies()
	{
		foreach (WeaponCategory allCategory in WeaponSubCategory.Natural.GetAllCategories())
		{
			if (!Proficiencies.Contains(allCategory))
			{
				Proficiencies.Add(allCategory);
			}
		}
	}

	public void Dispose()
	{
		if (Body != null && Body.Owner == this)
		{
			Body.Dispose();
		}
		Progression.Dispose();
		foreach (Spellbook value in m_Spellbooks.Values)
		{
			value.Dispose();
		}
		if (!IsPlayerFaction && Inventory.OwnerUnit == this)
		{
			Inventory.Dispose();
		}
	}

	public FactionSwitcher GetFactionSwitcher(BlueprintFaction faction, bool resetAttackFactions = false)
	{
		return new FactionSwitcher(this, faction, resetAttackFactions);
	}

	public void SwitchFactions(BlueprintFaction faction, bool resetAttackFactions = false)
	{
		SwitchFactions(faction, resetAttackFactions, permanently: true);
	}

	private void SwitchFactions(BlueprintFaction faction, bool resetAttackFactions, bool permanently)
	{
		if (Faction == faction)
		{
			if (resetAttackFactions)
			{
				AttackFactions.Match(faction.AttackFactions);
			}
			return;
		}
		if (resetAttackFactions)
		{
			AttackFactions.Match(faction.AttackFactions);
		}
		else
		{
			List<BlueprintFaction> second = AttackFactions.Except(Faction.AttackFactions).ToList();
			List<BlueprintFaction> second2 = Faction.AttackFactions.Except(AttackFactions).ToList();
			AttackFactions.Match(faction.AttackFactions.Except(second2).Concat(second));
		}
		m_FactionOriginal = (permanently ? faction : (m_FactionOriginal ?? Faction));
		Faction = faction;
		SetupInventory();
		EventBus.RaiseEvent(delegate(IUnitFactionHandler h)
		{
			h.HandleFactionChanged(Unit);
		});
		Unit.Group.UpdateAttackFactionsCache();
		if (Unit.IsInCombat && !Unit.Group.HasEnemyInCombat())
		{
			Unit.LeaveCombat();
		}
		foreach (UnitEntityData unit in Game.Instance.State.Units)
		{
			UnitPartSummonedMonster unitPartSummonedMonster = unit.Get<UnitPartSummonedMonster>();
			if (unitPartSummonedMonster != null && unitPartSummonedMonster.Summoner == Unit)
			{
				unitPartSummonedMonster.Owner.SwitchFactions(faction, resetAttackFactions);
			}
		}
	}

	public void FixInventoryOnPlayerPostLoad()
	{
		if (IsPlayerFaction && Unit.IsPet)
		{
			HasOwnInventory = false;
		}
		SetupInventory(restoreOnPlayerPostLoad: true);
	}

	private void SetupInventory(bool restoreOnPlayerPostLoad = false)
	{
		ItemsCollection itemsCollection = m_Inventory;
		if (IsPlayerFaction && !HasOwnInventory)
		{
			itemsCollection = Game.Instance.Player.EnsureInventory();
		}
		else if (m_Inventory?.OwnerUnit != this)
		{
			itemsCollection = new ItemsCollection((UnitEntityData)this);
		}
		if (itemsCollection == null)
		{
			PFLog.Default.Error("New inventory is null");
		}
		else
		{
			if (m_Inventory == itemsCollection)
			{
				return;
			}
			if (restoreOnPlayerPostLoad)
			{
				PFLog.Default.Error(OriginalBlueprint, $"Fix inventory for unit {this} (faction: {Faction})");
			}
			m_Inventory = itemsCollection;
			using (ContextData<ItemsCollection.DoNotRemoveFromSlot>.Request())
			{
				foreach (ItemEntity item in Body.Items)
				{
					if (item.Collection != null && item.Collection != itemsCollection)
					{
						PFLog.Default.Log($"{this}: move item {item} to actual inventory");
						item.Collection.Transfer(item, itemsCollection);
					}
				}
			}
		}
	}

	public void SetFakeBlueprint([CanBeNull] BlueprintUnit fake)
	{
		if (fake == null)
		{
			Blueprint = OriginalBlueprint;
		}
		else
		{
			Blueprint = fake;
		}
	}

	public UnitDescriptor(BlueprintUnit blueprint, UnitEntityData data)
	{
		UnitSpawningData current = ContextData<UnitSpawningData>.Current;
		Unit = data;
		Unit.Descriptor = this;
		Blueprint = (OriginalBlueprint = blueprint);
		Faction = blueprint.Faction;
		AttackFactions = new UnitAttackFactions(this);
		Stats = new CharacterStats(this);
		m_Inventory = current?.Inventory ?? (IsPlayerFaction ? Game.Instance.Player.EnsureInventory() : new ItemsCollection((UnitEntityData)this));
		Progression = new UnitProgressionData(this);
		Body = current?.Body ?? new UnitBody(this);
		Brain = new UnitBrain(this);
		Resources = new UnitAbilityResourceCollection(this);
		State = new UnitState(this);
		Alignment = new UnitAlignment(this);
		UISettings = new UnitUISettings(this);
		Blueprint.CallComponents(delegate(FixUnitOnPostLoad a)
		{
			if (!m_AppliedFixers.HasItem(a.name))
			{
				m_AppliedFixers = m_AppliedFixers ?? new List<string>();
				m_AppliedFixers.Add(a.name);
			}
		});
	}

	public void EnsureOwnInventory()
	{
		if (m_Inventory.IsPlayerInventory)
		{
			HasOwnInventory = true;
			SetupInventory();
		}
	}

	public void SetupSharedInventoryForPet()
	{
		if (IsPlayerFaction && !m_Inventory.IsPlayerInventory)
		{
			HasOwnInventory = false;
			SetupInventory();
		}
	}

	public void Initialize()

	}

	private static void Initialize(UnitDescriptor unit, BlueprintUnit blueprint)
	{
		if (blueprint == null)
		{
			PFLog.Default.Error("Unit created without blueprint");
			return;
		}
		unit.Ensure<LevelUpPlanUnitHolder>();
		unit.Stats.HitPoints.BaseValue = blueprint.MaxHP;
		InitializeStat(unit.Stats.Strength, blueprint.Strength);
		InitializeStat(unit.Stats.Dexterity, blueprint.Dexterity);
		InitializeStat(unit.Stats.Constitution, blueprint.Constitution);
		InitializeStat(unit.Stats.Intelligence, blueprint.Intelligence);
		InitializeStat(unit.Stats.Wisdom, blueprint.Wisdom);
		InitializeStat(unit.Stats.Charisma, blueprint.Charisma);
		unit.Stats.Speed.BaseValue = blueprint.Speed.Value;
		unit.SetSkills(blueprint.Skills);
		if (!unit.IsPlayerFaction)
		{
			unit.HasOwnInventory = true;
		}
		unit.EnsureNaturalProficiencies();
		unit.Alignment.Initialize(blueprint.Alignment);
		bool flag = true;
		bool flag2 = true;
		UnitSpawningData current = ContextData<UnitSpawningData>.Current;
		if (current != null)
		{
			if (current.Race != null)
			{
				unit.Progression.SetRace(current.Race);
			}
			if (current.Voice != null)
			{
				unit.CustomAsks = current.Voice;
			}
			if (current.Gender.HasValue)
			{
				unit.CustomGender = current.Gender;
			}
			if (current.LeftHanded.HasValue)
			{
				unit.LeftHandedOverride = current.LeftHanded;
			}
			if (!unit.Body.IsPolymorphed)
			{
				unit.CustomPrefabGuid = current.PrefabGuid;
			}
			flag = current.Inventory == null && !current.IsFromPlaceholder;
			flag2 = current.Body == null;
			current.Body?.SetOwner(unit);
			current.Inventory?.SetOwner(unit);
			current.Body?.PrePostLoad(unit);
		}
		if (unit.Progression.Race == null && blueprint.Race != null)
		{
			unit.Progression.SetRace(blueprint.Race);
		}
		unit.State.Size = unit.OriginalSize;
		if (!unit.Faction)
		{
			PFLog.Default.Error(blueprint, "Faction not set for " + blueprint);
		}
		unit.AttackFactions.UnionWith(blueprint.AttackFactions);
		unit.MainFact = unit.Unit.Facts.Add(new MainUnitFact(blueprint));
		unit.AddFact(BlueprintRoot.Instance.SystemMechanics.ChargeAbility);
		(from a in blueprint.GetComponents<AddAbilityToCharacterComponent>().SelectMany((AddAbilityToCharacterComponent c) => c.Abilities)
			where a
			select a).ForEach(delegate(BlueprintAbility a)
		{
			unit.AddFact(a);
		});
		(from a in blueprint.GetComponents<AddActivatableAbilityComponent>().SelectMany((AddActivatableAbilityComponent c) => c.ActivatableAbilities)
			where a
			select a).ForEach(delegate(BlueprintActivatableAbility a)
		{
			unit.AddFact(a);
		});
		foreach (BlueprintUnitFact item in blueprint.AddFacts.EmptyIfNull())
		{
			if ((bool)item)
			{
				unit.AddFact(item);
			}
		}
		foreach (BlueprintUnitTemplate item2 in blueprint.AdditionalTemplates.EmptyIfNull())
		{
			ApplyTemplate(unit, item2);
		}
		if (!ContextData<UnitEntityData.ChargenUnit>.Current)
		{
			if (flag2)
			{
				using (ProfileScope.New("Body"))
				{
					unit.Body.Initialize();
				}
			}
			if (flag)
			{
				using (ProfileScope.New("Starting Inventory"))
				{
					AddStartingInventory(unit, blueprint);
				}
			}
		}
		else
		{
			unit.Body.EnsureActiveHands();
		}
		using (ProfileScope.New("Brain"))
		{
			unit.Brain.RestoreAvailableActions();
		}
		using (ProfileScope.New("HandleInventoryInitialized Event"))
		{
			blueprint.CallComponents(delegate(IHandleInventoryInitialized c)
			{
				c.OnInventoryInitialized(unit);
			});
		}
		unit.Remove<LevelUpPlanUnitHolder>();
		if (unit.IsPlayerFaction)
		{
			unit.Ensure<UnitPartPartyWeatherBuff>();
		}
		unit.IsTurnedOn = true;
		foreach (Spellbook spellbook in unit.Spellbooks)
		{
			spellbook.UpdateAllSlotsSize(allowShrink: true);
		}
		unit.Replenish();
	}

	private static void AddStartingInventory(UnitDescriptor unit, BlueprintUnit blueprint)

		}
	}

	private static void InitializeStat(ModifiableValueAttributeStat stat, int value)

	public static void ApplyTemplate(UnitDescriptor unit, BlueprintUnitTemplate template)
	{
		if (template == null)
		{
			return;
		}
		foreach (BlueprintUnitFact item in template.AddFacts.EmptyIfNull())
		{
			if ((bool)item)
			{
				unit.AddFact(item);
			}
			else
			{
				PFLog.Default.Warning("NULL fact in template");
			}
		}
		foreach (BlueprintUnitFact item2 in template.RemoveFacts.EmptyIfNull())
		{
			if ((bool)item2)
			{
				unit.AddFact(item2);
			}
			else
			{
				PFLog.Default.Warning("NULL buff in template");
			}
		}
		foreach (BlueprintUnitTemplate.StatAdjustment item3 in template.StatAdjustments.EmptyIfNull())
		{
			ModifiableValue stat = unit.Stats.GetStat(item3.Stat);
			if (stat != null)
			{
				if (!(stat is ModifiableValueAttributeStat modifiableValueAttributeStat) || (modifiableValueAttributeStat.Enabled && modifiableValueAttributeStat.BaseValue > 2))
				{
					stat.BaseValue += item3.Adjustment;
				}
			}
			else
			{
				PFLog.Default.Warning($"Cannot change stat {item3.Stat}: stat not found", template);
			}
		}
	}

	[NotNull]
	public Spellbook DemandSpellbook([NotNull] BlueprintCharacterClass characterClass)
	{
		ClassData classData = Progression.GetClassData(characterClass);
		if (classData == null)
		{
			if (characterClass.Spellbook != null)
			{
				PFLog.Default.Warning($"Demanding spellbook for class unit doesnt have: {this}, {characterClass}. Returning default");
				return DemandSpellbook(characterClass.Spellbook);
			}
			PFLog.Default.Error($"Demanding spellbook for class unit doesnt have: {this}, {characterClass}. Class doesnt have default spellbook.");
			return new Spellbook(this, null);
		}
		if (classData.Spellbook == null)
		{
			PFLog.Default.Error($"Demanding spellbook for class without one: {this}, {characterClass}.");
			return new Spellbook(this, null);
		}
		return DemandSpellbook(classData.Spellbook);
	}

	private void Resurrect([CanBeNull] UnitEntityData initiator, float resultHealthPart, bool restoreHealth, bool fullRestore)
	{
		bool isFinallyDead = State.IsFinallyDead;
		State.IsFinallyDead = false;
		State.RemoveConditionAll(UnitCondition.DeathDoor);
		State.RemoveConditionAll(UnitCondition.WalkingDead);
		State.ForceKill = false;
		State.MarkedForDeath = false;
		if (fullRestore)
		{
			Damage = 0;
		}
		if (Damage > (int)Stats.HitPoints - Progression.CharacterLevel)
		{
			Damage = (restoreHealth ? Math.Min((int)Stats.HitPoints - Math.Max(1, Progression.CharacterLevel), (int)((float)(int)Stats.HitPoints * (1f - resultHealthPart))) : Damage);
		}
		StatType[] attributes = StatTypeHelper.Attributes;
		foreach (StatType type in attributes)
		{
			ModifiableValueAttributeStat stat = Stats.GetStat<ModifiableValueAttributeStat>(type);
			if (fullRestore)
			{
				stat.Damage = 0;
				stat.Drain = 0;
			}
			if (stat.ModifiedValueRaw < 1)
			{
				int num = -stat.ModifiedValueRaw + 1;
				int num2 = Math.Min(stat.Damage, num);
				stat.Damage -= num2;
				num -= num2;
				int num3 = Math.Min(stat.Drain, num);
				stat.Drain -= num3;
			}
		}
		UnitPartNegativeLevels unitPartNegativeLevels = Get<UnitPartNegativeLevels>();
		if ((bool)unitPartNegativeLevels && fullRestore)
		{
			unitPartNegativeLevels.RemoveAll();
		}
		while ((bool)unitPartNegativeLevels && unitPartNegativeLevels.Count >= Progression.CharacterLevel)
		{
			if (unitPartNegativeLevels.RemoveTemporary(all: false) < 1 && unitPartNegativeLevels.RemovePermanent(all: false) < 1)
			{
				PFLog.Default.Error("Can't remove negative levels, something wrong");
				break;
			}
		}
		if ((bool)Unit.View)
		{
			if (IsPlayerFaction)
			{
				initiator = initiator ?? Game.Instance.Player.GetMainPartyUnit();
				if (Unit.IsHiddenBecauseDead)
				{
					Vector3 resurrectPosition = ResurrectionLogic.GetResurrectPosition(initiator, Unit);
					Unit.Position = resurrectPosition;
					Unit.View.gameObject.transform.position = resurrectPosition;
				}
				else if (isFinallyDead && AstarPath.active.GetNearest(Unit.Position).node?.Area != AstarPath.active.GetNearest(initiator.Position).node?.Area)
				{
					Vector3 resurrectPosition2 = ResurrectionLogic.GetResurrectPosition(initiator, Unit);
					Unit.Position = resurrectPosition2;
					Unit.View.gameObject.transform.position = resurrectPosition2;
				}
			}
			Unit.View.RigidbodyController?.CancelRagdoll();
			State.Prone.Active = false;
			Unit.View.LeaveProneState();
			Unit.View.AnimationManager?.StandUpImmediately();
			Game.Instance.HandsEquipmentController?.ScheduleUpdate(Unit);
			Unit.View.ResetMouseHighlighted();
		}
		Unit.IsHiddenBecauseDead = false;
		UnitPartDualCompanion unitPartDualCompanion = Get<UnitPartDualCompanion>();
		if (unitPartDualCompanion != null)
		{
			unitPartDualCompanion.IsDead = false;
		}
		RemoveBuffsOnResurrect();
		EventBus.RaiseEvent(delegate(IUnitResurrectedHandler h)
		{
			h.HandleUnitResurrected(Unit);
		});
		if (State.IsConscious)
		{
			return;
		}
		foreach (Buff buff in Buffs)
		{
			foreach (AddCondition item in buff.SelectComponents<AddCondition>())
			{
				if (item.Condition == UnitCondition.Petrified)
				{
					buff.Remove();
					break;
				}
			}
			foreach (BuffStatusCondition item2 in buff.SelectComponents<BuffStatusCondition>())
			{
				if (item2.Condition == UnitCondition.Petrified)
				{
					buff.Remove();
					break;
				}
			}
		}
		State.RemoveConditionAll(UnitCondition.Petrified);
		UnitLifeController.ForceUnitConscious(this);
	}

	public void RemoveBuffsOnResurrect()
	{
		List<Buff> list = null;
		foreach (Buff buff in Buffs)
		{
			if (buff.Blueprint.RemoveOnResurrect)
			{
				list = list ?? new List<Buff>();
				list.Add(buff);
			}
		}
		if (list == null)
		{
			return;
		}
		foreach (Buff item in list)
		{
			Buffs.RemoveFact(item);
		}
	}

	public void UpdateSizeModifiers()
	{
		Stats.AdditionalAttackBonus.RemoveModifiers(ModifierDescriptor.Size);
		Stats.AC.RemoveModifiers(ModifierDescriptor.Size);
		Stats.SkillStealth.RemoveModifiers(ModifierDescriptor.Size);
		Stats.Reach.RemoveModifiers(ModifierDescriptor.Size);
		ObjectExtensions.Or(Unit.View, null)?.ResetSizeScale();
		WeaponSizeExtension.SizeModifiers modifiers = State.Size.GetModifiers();
		if (modifiers != null)
		{
			Stats.AdditionalAttackBonus.AddModifier(modifiers.AttackAndAC, ModifierDescriptor.Size);
			Stats.AC.AddModifier(modifiers.AttackAndAC, ModifierDescriptor.Size);
			Stats.SkillStealth.AddModifier(modifiers.Stealth, ModifierDescriptor.Size);
			Stats.Reach.AddModifier(modifiers.Reach, ModifierDescriptor.Size);
		}
	}

	public Feet GetWeaponRange([CanBeNull] BlueprintItemWeapon weapon = null)
	{
		Feet feet = SimpleBlueprintExtendAsObject.Or(weapon, null)?.AttackRange ?? GameConsts.MinWeaponRange;
		if ((bool)State.Features.PointBlankRange && feet.Value > 30)
		{
			feet = 30.Feet();
		}
		Feet reachRange = Stats.ReachRange;
		return feet + reachRange;
	}

	public void OnDifficultyChanged()
	{
		if (IsPlayerFaction)
		{
			return;
		}
		foreach (Feature feature in Progression.Features)
		{
			if ((bool)feature.Context.AssociatedBlueprint.GetComponent<ContextAbilityParamsCalculator>())
			{
				feature.Context.RecalculateAbilityParams();
			}
		}
		foreach (Buff buff in Buffs)
		{
			if ((bool)buff.Context.AssociatedBlueprint.GetComponent<ContextAbilityParamsCalculator>())
			{
				buff.Context.RecalculateAbilityParams();
			}
		}
	}
