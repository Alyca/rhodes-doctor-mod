	public static void KillUnit(UnitEntityData unit, UnitEntityData killer = null, bool byCrit = false, bool noForce = false)
	{
		if (noForce && !TacticalCombatHelper.IsActive)
		{
			unit.Descriptor.State.MarkedForDeath = true;
		}
		else
		{
			unit.Descriptor.State.ForceKill = true;
		}
		unit.Wake(1f);
		if (killer != null)
		{
			unit.LastHandledDamage = new RuleDealDamage(killer, unit, new DirectDamage(DiceFormula.Zero)
			{
				CriticalModifier = (byCrit ? new int?(DamageCriticalModifierType.X4.IntValue()) : ((int?)null))
			});
		}
		else
		{
			unit.LastHandledDamage = null;
		}
	}

	public static bool CheckStatResult(UnitEntityData unit, StatType statType, int difficultyClass, RuleSkillCheck.VoicingType voice = RuleSkillCheck.VoicingType.All, bool? ensureSuccess = null, bool allowPartyCheckInCamp = true)
	{
		return TriggerStatCheck(RuleStatCheck.Create(unit, statType, difficultyClass, voice, ensureSuccess), null, allowPartyCheckInCamp).Success;
	}

	public static bool CheckPartySkillResult(StatType statType, int difficultyClass, RuleSkillCheck.VoicingType voice = RuleSkillCheck.VoicingType.All, bool? ensureSuccess = null, bool allowPets = false)
	{
		RulePartySkillCheck obj = new RulePartySkillCheck(statType, difficultyClass, applyPartyChecksInCapital: true, allowPets)
		{
			Voice = voice,
			EnsureSuccess = ensureSuccess
		};
		Rulebook.Trigger(obj);
		return obj.Success;
	}

	public static RuleSkillCheck TriggerSkillCheck(RuleSkillCheck skillCheck, MechanicsContext context = null, bool allowPartyCheckInCamp = true)
	{
		if (allowPartyCheckInCamp && Game.Instance.Player.CapitalPartyMode)
		{
			RulePartySkillCheck rulePartySkillCheck = new RulePartySkillCheck(skillCheck);
			TriggerRule(rulePartySkillCheck, context);
			return rulePartySkillCheck.SkillCheck;
		}
		return TriggerRule(skillCheck, context);
	}

	public static RuleStatCheck TriggerStatCheck(RuleStatCheck statCheck, MechanicsContext context = null, bool allowPartyCheckInCamp = true)
	{
		if (allowPartyCheckInCamp && Game.Instance.Player.CapitalPartyMode)
		{
			RulePartyStatCheck rulePartyStatCheck = new RulePartyStatCheck(statCheck);
			TriggerRule(rulePartyStatCheck, context);
			return rulePartyStatCheck.StatCheck;
		}
		return TriggerRule(statCheck);
	}

	public static TEvent TriggerRule<TEvent>(TEvent rule, MechanicsContext context = null) where TEvent : RulebookEvent
	{
		if (context != null)
		{
			return context.TriggerRule(rule);
		}
		return Rulebook.Trigger(rule);
	}

		return GetTargetsAround(point, radius.Meters, checkLOS, includeDead);

	public static IEnumerable<UnitEntityData> GetTargetsAround(Vector3 point, float radius, bool checkLOS = true, bool includeDead = false)
	{
		foreach (UnitEntityData unit in Game.Instance.State.Units)
		{
			if ((!unit.Descriptor.State.IsDead || includeDead) && !unit.Descriptor.State.Features.IsUntargetable && unit.IsUnitInRange(point, radius, checkLOS))
			{
				yield return unit;
			}
		}
	}

	public static bool IsUnitInRange(this UnitEntityData unit, Vector3 point, float radius, bool checkLOS = true)
	{
		float num = unit.DistanceTo(point);
		float num2 = radius + unit.Corpulence;
		if (num <= num2)
		{
			if (checkLOS)
			{
				return CheckLOS(point, unit.Position);
			}
			return true;
		}
		return false;
	}

	public static int GetItemEnhancementBonus(ItemEntityWeapon weapon)
	{
		List<WeaponEnhancementBonus> source = weapon.Enchantments.SelectMany((ItemEnchantment p) => p.SelectComponents<WeaponEnhancementBonus>()).ToTempList();
		List<WeaponEnhancementBonus> source2 = source.Where((WeaponEnhancementBonus p) => p.Stack).ToTempList();
		List<WeaponEnhancementBonus> source3 = source.Where((WeaponEnhancementBonus p) => !p.Stack).ToTempList();
		int num = ((!source3.Empty()) ? source3.Max((WeaponEnhancementBonus p) => p.EnhancementBonus) : 0);
		int num2 = ((!source2.Empty()) ? source2.Sum((WeaponEnhancementBonus p) => p.EnhancementBonus) : 0);
		if (num < 5)
		{
			return Math.Min(num + num2, 5);
		}
		return num;
	}

	public static int GetItemEnhancementBonus(ItemEntity item)
	{
		if (item is ItemEntityWeapon weapon)
		{
			GetItemEnhancementBonus(weapon);
		}
		ItemEntityArmor itemEntityArmor = (item as ItemEntityArmor) ?? (item as ItemEntityShield)?.ArmorComponent;
		if (itemEntityArmor != null)
		{
			int? num = null;
			foreach (ItemEnchantment enchantment in itemEntityArmor.Enchantments)
			{
				ArmorEnhancementBonus component = enchantment.GetComponent<ArmorEnhancementBonus>();
				if (component != null && (!num.HasValue || component.EnhancementValue > num))
				{
					num = component.EnhancementValue;
					if ((bool)itemEntityArmor.Wielder?.State.Features.AdvancedFakeEnhancement)
					{
						num += 2;
					}
					else if ((bool)itemEntityArmor.Wielder?.State.Features.FakeEnhancement)
					{
						num++;
					}
				}
			}
			return num.GetValueOrDefault();
		}
		return 0;
	}

	public static int GetArmorEnhancementBonus(BlueprintItemArmor blueprint)
	{
		int? num = null;
		foreach (BlueprintItemEnchantment enchantment in blueprint.Enchantments)
		{
			ArmorEnhancementBonus component = enchantment.GetComponent<ArmorEnhancementBonus>();
			if (component != null && (!num.HasValue || component.EnhancementValue > num))
			{
				num = component.EnhancementValue;
			}
		}
		return num.GetValueOrDefault();
	}

	public static int GetWeaponEnhancementBonus(BlueprintItemWeapon weapon)
	{
		List<WeaponEnhancementBonus> source = weapon.Enchantments.SelectMany((BlueprintItemEnchantment p) => p.GetComponents<WeaponEnhancementBonus>()).ToList();
		List<WeaponEnhancementBonus> source2 = source.Where((WeaponEnhancementBonus p) => p.Stack).ToList();
		List<WeaponEnhancementBonus> source3 = source.Where((WeaponEnhancementBonus p) => !p.Stack).ToList();
		int num = (source3.Any() ? source3.Max((WeaponEnhancementBonus p) => p.EnhancementBonus) : 0);
		int num2 = (source2.Any() ? source2.Sum((WeaponEnhancementBonus p) => p.EnhancementBonus) : 0);
		if (num < 5)
		{
			return Math.Min(num + num2, 5);
		}
		return num;
	}

	public static void GainExperience(int gained, UnitEntityData unit = null, ExperienceGainStatistic.GainType statType = ExperienceGainStatistic.GainType.Quest)
	{
		unit = unit?.Master ?? unit;
		if (unit != null && (bool)SettingsRoot.Difficulty.OnlyActiveCompanionsReceiveExperience)
		{
			unit.Descriptor.Progression.GainExperience(gained * 6, log: true, allowForPet: false, statType);
		}
		else
		{
			Game.Instance.Player.GainPartyExperience(gained, statType);
		}
	}

	public static void GainExperienceForSkillCheck(int gained, [NotNull] UnitEntityData unit, ExperienceGainStatistic.GainType statType = ExperienceGainStatistic.GainType.Check)
	{
		unit = unit.Master ?? unit;

		{
			unit.Descriptor.Progression.GainExperience(gained * 6, log: true, allowForPet: false, statType);
		}
		else
		{

		}
	}
