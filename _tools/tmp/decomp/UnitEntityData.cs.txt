	private float ConsoleSpeedModifier
	{
		get
		{

	public Feet NonCombatBaseSpeedFeet
	{
		get
		{
			if (IsPlayerFaction && (bool)SettingsRoot.Difficulty.EncumbranceSlowdown && Game.Instance.CurrentMode != GameModeType.Cutscene)
			{
				if (Game.Instance.Player.Encumbrance == Encumbrance.Heavy)
				{
					return 20.Feet() * ConsoleSpeedModifier;
				}
				if (Game.Instance.Player.Encumbrance == Encumbrance.Overload)
				{
					return 10.Feet() * ConsoleSpeedModifier;
				}
			}
			return 30.Feet() * ConsoleSpeedModifier;
		}
	}

	public float BaseSpeedMps
	{
		get
		{
			UnitEntityData saddledUnit = this.GetSaddledUnit();
			if (saddledUnit != null && Blueprint.GetArmyData() == null)
			{
				return saddledUnit.BaseSpeedMps;
			}
			if (BuildModeUtility.IsCheatsEnabled && !CheatsAnimation.SpeedLock)
			{
				return Stats.Speed.Mps;
			}
			if (IsPlayerFaction && !IsInCombat)
			{
				Feet nonCombatBaseSpeedFeet = NonCombatBaseSpeedFeet;
				float num = nonCombatBaseSpeedFeet.Meters / 3f;
				if (nonCombatBaseSpeedFeet < 30.Feet())
				{
					return num;
				}
				return Mathf.Max(num, Stats.Speed.Mps);
			}
			return Stats.Speed.Mps;
		}
	}

	public float CurrentSpeedMps
	{
		get
		{
			if (!m_MemoizedSpeed.UpToDate || TacticalCombatHelper.IsActive)
			{
				m_MemoizedSpeed.Value = CalculateCurrentSpeed();
			}
			return m_MemoizedSpeed.Value;
		}
	}

	public float CombatSpeedMps
	{
		get
		{
			UnitEntityData unitEntityData = Get<UnitPartRider>()?.SaddledUnit;
			if (unitEntityData != null && Blueprint.GetArmyData() == null)
			{
				return unitEntityData.BaseSpeedMps;
			}
			return Math.Max(GameConsts.MinUnitSpeedMps, Stats.Speed.Mps * CalculateSpeedModifier());
		}
	}

	public float ModifiedSpeedMps
	{
		get
		{
			UnitEntityData unitEntityData = Get<UnitPartRider>()?.SaddledUnit;
			if (unitEntityData != null && Blueprint.GetArmyData() == null)
			{
				return unitEntityData.ModifiedSpeedMps;
			}
			return Math.Max(GameConsts.MinUnitSpeedMps, BaseSpeedMps * CalculateSpeedModifier());
		}
	}

	public bool IsDirectlyControllable
	{
		get
		{
			if (!base.IsInGame)
			{
				return false;
			}
			if ((bool)PreventDirectControl)
			{
				return false;
			}
			CountingGuard countingGuard = Game.Instance.LoadedAreaState?.Settings.CapitalPartyMode;
			if (countingGuard != null && (bool)countingGuard)
			{
				if (IsMainCharacter)
				{
					return true;
				}
				UnitEntityData master = Master;
				if ((object)master != null && master.IsMainCharacter)
				{
					CompanionSpawner companionSpawner = Get<UnitPartCompanion>()?.GetCurrentSpawner();
					if (!(companionSpawner == null))
					{
						return companionSpawner.SpawnedUnit != this;
					}
					return true;
				}
				return false;
			}
			if (TacticalCombatHelper.IsActive)
			{
				UnitPartLeaderTacticalCombat unitPartLeaderTacticalCombat = Get<UnitPartLeaderTacticalCombat>();
				if (unitPartLeaderTacticalCombat == null || unitPartLeaderTacticalCombat.Faction != ArmyFaction.Crusaders)
				{
					UnitPartTacticalCombat unitPartTacticalCombat = Get<UnitPartTacticalCombat>();
					if (unitPartTacticalCombat == null || unitPartTacticalCombat.Faction != ArmyFaction.Crusaders)
					{
						return false;
					}
				}
				return !Descriptor.State.IsFinallyDead;
			}
			if (!Descriptor.Faction.IsDirectlyControllable || Descriptor.State.IsFinallyDead || Descriptor.State.IsPanicked || IsDetached)
			{
				return false;
			}
			if (Get<UnitPartSummonedMonster>() != null && !Blueprint.Faction.IsDirectlyControllable)
			{
				return false;
			}
			UnitPartCompanion unitPartCompanion = Master?.Get<UnitPartCompanion>() ?? Get<UnitPartCompanion>();
			if (unitPartCompanion != null && unitPartCompanion.State != CompanionState.ExCompanion)
			{
				return unitPartCompanion.State != CompanionState.Remote;
			}
			return false;
		}
	}

	private float CalculateCurrentSpeed()
	{
		UnitEntityData unitEntityData = Get<UnitPartRider>()?.SaddledUnit;
		if (unitEntityData != null && Blueprint.GetArmyData() == null)
		{
			return unitEntityData.CurrentSpeedMps;
		}
		UnitCommand unitCommand = Commands.GetCommand(UnitCommand.CommandType.Move) ?? Commands.Standard;
		if (unitCommand != null && unitCommand.OverrideSpeed.HasValue)
		{
			return unitCommand.OverrideSpeed.Value;
		}
		float num = ModifiedSpeedMps;
		if (unitCommand != null)
		{
			switch (unitCommand.MovementType)
			{
			case UnitAnimationActionLocoMotion.WalkSpeedType.Slow:
				num *= 0.5f;
				break;
			case UnitAnimationActionLocoMotion.WalkSpeedType.Charge:
				if (unitCommand != null && unitCommand.IsAccelerated)
				{
					if (View?.AgentOverride == null)
					{
						num *= 1.8f;
					}
				}
				else
				{
					num *= 1.2f;
				}
				break;
			case UnitAnimationActionLocoMotion.WalkSpeedType.Stealth:
				num *= 0.5f;
				break;
			default:
				throw new ArgumentOutOfRangeException();
			case UnitAnimationActionLocoMotion.WalkSpeedType.Normal:
				break;
			}
			if ((!IsInCombat || unitCommand.ApplySpeedLimitInCombat) && unitCommand.SpeedLimit.HasValue && unitCommand.SpeedLimit < num)
			{
				num = unitCommand.SpeedLimit.Value;
			}
		}
		num = Math.Max(GameConsts.MinUnitSpeedMps, num);
		if (!IsInCombat && Game.Instance.IsControllerGamepad && IsDirectlyControllable)
		{
			UnitEntityView unitEntityView = Game.Instance.SelectionCharacter.SingleSelectedUnit?.View;
			UnitMovementAgentContinuous unitMovementAgentContinuous = unitEntityView?.AgentOverride as UnitMovementAgentContinuous;
			if (unitEntityView != null && this != unitEntityView.EntityData && this != unitEntityView.EntityData.GetSaddledUnit() && unitEntityView.EntityData.Commands.MoveContinuously != null && unitMovementAgentContinuous != null)
			{
				num *= unitMovementAgentContinuous?.DirectionFromControllerMagnitude ?? 1f;
				if (num < unitMovementAgentContinuous.Speed)
				{
					return num;
				}
				return unitMovementAgentContinuous.Speed;
			}
		}
		return num;
	}

	public void UpdateHasDummyView()
	{
		if (HoldingState == null)
		{
			return;
		}
		bool flag = HoldingState != Game.Instance.Player.CrossSceneState && (base.Suppressed || ((bool)Get<UnitPartOptimizeWhenHidden>() && !base.IsInGame));
		StartupJson data = BuildModeUtility.Data;
		if (data != null && data.DisableUnitOptimization)
		{
			flag = false;
		}
		if ((!flag && IsSleeping && HasDummyView && !IsDeathRevealed) || flag == HasDummyView)
		{
			return;
		}
		HasDummyView = flag;
		UnitEntityView view = View;
		if (!view)
		{
			return;
		}
		using (ProfileScope.New("Change View Dummy-to-Real"))
		{
			AttachToViewOnLoad(null);
			Game.Instance.DynamicRoot.Add(View.transform);
			using (ProfileScope.New("Update View state"))
			{
				UnitAnimationManager animationManager = View.AnimationManager;
				if (animationManager != null)
				{
					animationManager.Tick();
					animationManager.UpdateTime(0.1f);
				}
				using (ProfileScope.New("Update Visibility"))
				{
					if ((bool)View.Fader)
					{
						View.SetVisible(View.IsVisible, force: true);
					}
				}
			}
			UnityEngine.Object.Destroy(view.gameObject);
		}
	}

	protected override void OnIsInGameChanged()
	{
		using (ProfileScope.New("base.OnIsInGameChanged"))
		{
			base.OnIsInGameChanged();
		}
		using (ProfileScope.New("UpdateHasDummyView"))
		{
			UpdateHasDummyView();
		}
		if ((bool)Parts.Get<UnitPartCompanion>())
		{
			Game.Instance.Player.InvalidateCharacterLists();
		}
		if (base.IsInGame)
		{
			TranslocatePetsClose();
		}
		foreach (EntityPartRef<UnitEntityData, UnitPartPet> pet in Pets)
		{
			if (pet.Entity != null && pet.EntityPart != null && (!pet.EntityPart.ShouldBeHidden || !base.IsInGame))
			{
				pet.Entity.IsInGame = base.IsInGame;
			}
		}
		using (ProfileScope.New("Buffs de/activate"))
		{
			Descriptor.Buffs.SetActive(base.IsInGame);
		}
		using (ProfileScope.New("ActAbs de/activate"))
		{
			Descriptor.ActivatableAbilities.SetActive(base.IsInGame);
		}
		using (ProfileScope.New("Abilities de/activate"))
		{
			Descriptor.Abilities.SetActive(base.IsInGame);
		}
		m_Group?.Remove(this);
		m_Group = null;
		LastHandledDamage = null;
	}

	private void TranslocatePetsClose()
	{
		if (!AstarPath.active)
		{
			return;
		}
		foreach (EntityPartRef<UnitEntityData, UnitPartPet> pet in Pets)
		{
			UnitEntityData entity = pet.Entity;
			if (!(entity == null))
			{
				float num = Corpulence + entity.Corpulence;
				if (DistanceTo(entity) > num * 2f || ObstacleAnalyzer.GetArea(Position) != ObstacleAnalyzer.GetArea(entity.Position))
				{
					Vector3 vector = UnityEngine.Random.insideUnitCircle.normalized.To3D() * num * 2f;
					Vector3 pos = Position + vector;
					pos = ObstacleAnalyzer.GetNearestNode(pos).position;
					entity.Position = pos;
				}
			}
		}
	}

	public void DropLootToGround(bool dismember = false, Vector3? overridePos = null, bool force = false)

	public bool HasLOS(Vector3 point, Vector3 overridePosition)
	{
		try
		{
			Vector3 p = overridePosition + LineOfSightGeometry.EyeShift;
			CheckLOSBetweenPoints(p, point, out var hasLOS, out var _);
			return hasLOS;
		}
		finally
		{
		}
	}

	public void CheckLOS(Vector3 point, Vector3 overridePosition, out bool hasLOS, out bool hasLOSWithoutDistance)
	{
		try
		{
			Vector3 p = overridePosition + LineOfSightGeometry.EyeShift;
			CheckLOSBetweenPoints(p, point, out hasLOS, out hasLOSWithoutDistance);
		}
		finally
		{
		}
	}

	private void CheckLOSBetweenPoints(Vector3 p1, Vector3 p2, out bool hasLOS, out bool hasLOSWithoutDistance)
	{
		float visionRangeMeters = VisionRangeMeters;
		bool flag = GeometryUtils.SqrDistance2D(p1, p2) <= visionRangeMeters * visionRangeMeters;
		bool flag2 = LineOfSightGeometry.Instance.HasObstacle(p1, p2);
		bool flag3 = (object)ExtendedVisionArea != null && ExtendedVisionArea.ContainsPosition(p2);
		hasLOS = (flag && !flag2) || flag3;
		hasLOSWithoutDistance = !flag2 || flag3;
	}

	}

	public void JoinCombat(bool notSurprised)
	{
		UnitCommand standard = Commands.Standard;
		if (standard != null && !Game.Instance.Player.IsTurnBasedModeOn() && standard.IsActed && !standard.IsFinished && (!(standard is UnitAttack { IsAttackFull: not false } unitAttack) || unitAttack.GetAttackIndex() == 0))
		{
			return;

		CutsceneControlledUnit.UpdateActiveCutscene(this);
		EventBus.RaiseEvent(delegate(IUnitCombatHandler h)

		});

			if (View.gameObject.activeSelf)
			{
				UpdateCombatSwitch();

		}

	}

	public void LeaveCombat()
	{
		if (TacticalCombatHelper.IsActive && !State.IsFinallyDead && !base.Destroyed)
		{
			return;
		}
		CombatState.LeaveCombat();
		if (base.Destroyed)
		{
			return;
		}
		CancelStop();
		Commands.InterruptAiCommands();
		CutsceneControlledUnit.UpdateActiveCutscene(this);
		if ((bool)View && (bool)View.Animator)
		{
			View.HandsEquipment.OnCombatStateChanged(inCombat: false);
			if (View.gameObject.activeSelf)
			{
				UpdateCombatSwitch();
			}
		}
		if ((bool)SettingsRoot.Difficulty.RemoveAnnoyingBuffsAfterCombat)
		{
			Buffs.RemoveAnnoyingBuffs();
		}
		EventBus.RaiseEvent(delegate(IUnitCombatHandler h)
		{
			h.HandleUnitLeaveCombat(this);
		});
		PFLog.Default.Log(View, "Unit leave combat: {0}", this);
	}

	public UnitEntityView CreateView()
	{
		BlueprintComponentAndRuntime<Polymorph> activePolymorph = GetActivePolymorph();
		if (activePolymorph.Component != null)
		{
			UnitEntityView unitEntityView = activePolymorph.Component.CreatePolymorphView(this);
			if (unitEntityView != null)
			{
				unitEntityView.transform.position = Position;
				unitEntityView.transform.rotation = (unitEntityView.ForbidRotation ? Quaternion.identity : Quaternion.Euler(0f, Orientation, 0f));
				activePolymorph.GetData<PolymorphData>().ViewReplacement = unitEntityView.gameObject;
				unitEntityView.DisableSizeScaling = true;
				return unitEntityView;
			}
		}
		DollData dollData = Get<UnitPartDollData>()?.Default;
		if (dollData?.RacePreset != null)
		{
			UnitEntityView unitEntityView2 = dollData.CreateUnitView();
			unitEntityView2.transform.position = Position;
			unitEntityView2.transform.rotation = Quaternion.Euler(0f, Orientation, 0f);
			return unitEntityView2;
		}
		string text = (string.IsNullOrEmpty(Descriptor.CustomPrefabGuid) ? Blueprint.Prefab.AssetId : Descriptor.CustomPrefabGuid);
		text = (HasDummyView ? BlueprintRoot.Instance.OptimizationDummyUnit.AssetId : text);
		using (BundledResourceHandle<UnitEntityView> bundledResourceHandle = BundledResourceHandle<UnitEntityView>.Request(text))
		{
			UnitEntityView @object = bundledResourceHandle.Object;
			if ((bool)@object)
			{
				return InstantiateView_Internal(@object);
			}
		}
		if (OverrideLegacyUnits.Instance.TryGetOverrideGuid(text, out var resultGuid))
		{
			text = resultGuid;
			using BundledResourceHandle<UnitEntityView> bundledResourceHandle2 = BundledResourceHandle<UnitEntityView>.Request(text);
			UnitEntityView object2 = bundledResourceHandle2.Object;
			if ((bool)object2)
			{
				return InstantiateView_Internal(object2);
			}
		}
		PFLog.Default.Error(Blueprint, $"Cannot find prefab for unit {this} ({text})");
		return null;
	}

	private UnitEntityView InstantiateView_Internal(UnitEntityView originalPrefab)
	{
		Quaternion rotation = (originalPrefab.ForbidRotation ? Quaternion.identity : Quaternion.Euler(0f, Orientation, 0f));
		return UnityEngine.Object.Instantiate(originalPrefab, Position, rotation);
	}

			}
		}
		UnitUpgraderHelper.ApplyUpgraders(this, OriginalBlueprint, fromPlaceholder: false, ref m_AppliedUpgraders);
		foreach (EntityFact item in Facts.List)
		{
			if (item.Blueprint is BlueprintBuff { Stacking: StackingType.Replace } blueprintBuff)
			{
				m_BuffsDuplicates[blueprintBuff] = item;
			}
		}
		Facts.RemoveAll((Buff fact) => m_BuffsDuplicates.TryGetValue(fact.Blueprint, out var value) && value != fact);
		m_BuffsDuplicates.Clear();
	}

	protected override void OnTurnOn()
	{
		Descriptor.TurnOn();
		if (base.IsInGame && PreviewOf == null)
		{
			foreach (EntityFact item in Facts.List)
			{
				if (item.IsActive != item.IsEnabled)
				{
					PFLog.Default.ErrorWithReport(item.Blueprint, $"{item} enabled but not active, fixed");
				}
				item.UpdateIsActive();
			}
		}
		if (Brain.AutoUseAbility?.Fact != null && !Brain.AutoUseAbility.Fact.Active)
		{
			Ability ability = Abilities.Enumerable.FirstOrDefault((Ability a) => a.Blueprint == Brain.AutoUseAbility.Blueprint && a.SourceItem == Brain.AutoUseAbility.SourceItem);
			if (ability != null)
			{
				Brain.AutoUseAbility = ability.Data;
			}
		}
	}

	public void TransferInventoryToDroppedLoot()
	{
		if (m_DroppedLootEntityDataRef == null)
		{
			return;
		}
		DroppedLoot.EntityData entityData = m_DroppedLootEntityDataRef.Entity;
		if (entityData == null)
		{
			LogChannel.Default.Warning("Invalid DroppedLoot reference  " + m_DroppedLootEntityDataRef.Id + " from Unit " + base.UniqueId);
			return;
		}
		entityData.Loot = new ItemsCollection(entityData);
		using (ContextData<ItemSlot.IgnoreLock>.Request())
		{
			Inventory.Items.Where((ItemEntity i) => i.IsLootable && !i.NeedSkinningForCollect).ToArray().ForEach(delegate(ItemEntity i)
			{
				Inventory.Transfer(i, entityData.Loot);
			});
		}
		m_DroppedLootEntityDataRef = null;
		entityData.DroppedBy = null;
	}

	public float GetLookAtAngle(Vector3 point)
	{
		point.y = Position.y;
		if (point == Position)
		{
			return DesiredOrientation;
		}
		return Quaternion.LookRotation(point - Position).eulerAngles.y;
	}

		}

		}
	}

	private float CalculateSpeedModifier()
	{
		UnitEntityData saddledUnit = this.GetSaddledUnit();
		if (saddledUnit != null)
		{
			return saddledUnit.CalculateSpeedModifier();
		}
		float num = 1f;
		if ((bool)Game.Instance.LoadedAreaState.Settings.CapitalPartyMode)
		{
			return 1f;
		}
		if (Descriptor.State.IsInStealth && !Stealth.FullSpeed && !Descriptor.State.HasCondition(UnitCondition.Invisible))
		{
			num *= 0.5f;
		}
		if (Descriptor.State.HasCondition(UnitCondition.Staggered))
		{
			num *= 0.5f;
		}
		if (Descriptor.State.HasCondition(UnitCondition.Slowed))
		{
			num *= 0.5f;
		}
		if (Descriptor.State.HasCondition(UnitCondition.Entangled))
		{
			num *= 0.5f;
		}
		if (Descriptor.State.HasCondition(UnitCondition.DifficultTerrain))
		{
			num *= 0.5f;
		}
		if (Descriptor.State.HasCondition(UnitCondition.UseMobilityToNegateAttackOfOpportunity))
		{
			num *= (Descriptor.State.Features.TricksterMobilityFastMovement ? 1.5f : 0.5f);
		}
		if (BlueprintRoot.Instance.WeatherSettings.UseSlowdown && Game.Instance.Player.Weather.ActualWeather >= BlueprintRoot.Instance.WeatherSettings.SlowdownBonusBeginsOn)
		{
			num *= 0.5f;
		}
		return num;
	}

	[CanBeNull]
	public ItemEntityWeapon GetFirstWeapon()
	{
		if (IsNotUnarmedWeapon(this, Body.PrimaryHand))
		{
			return Body.PrimaryHand.MaybeWeapon;
		}
		ItemEntityWeapon maybeWeapon = Body.PrimaryHand.MaybeWeapon;
		if ((maybeWeapon == null || !maybeWeapon.Blueprint.IsTwoHanded) && IsNotUnarmedWeapon(this, Body.SecondaryHand))
		{
			return Body.SecondaryHand.MaybeWeapon;
		}
		foreach (WeaponSlot additionalLimb in Body.AdditionalLimbs)
		{
			if (IsNotUnarmedWeapon(this, additionalLimb))
			{
				return additionalLimb.MaybeWeapon;
			}
		}
		return null;
		static bool IsNotUnarmedWeapon(UnitEntityData unit, WeaponSlot hand)
		{
			if (hand.HasWeapon)
			{
				if (hand.Weapon.Blueprint.IsUnarmed)
				{
					return unit.Descriptor.State.Features.ImprovedUnarmedStrike;
				}
				return true;
			}
			return false;
		}
	}

	void IEntityPartsManagerDelegate.OnPartsManagerPostLoad(EntityPartsManager manager)
	{
		UnitPartInteractions.SetupBlueprintInteractions(this);
		UnitPartRider unitPartRider = Get<UnitPartRider>();
		ActivatableAbility fact = ActivatableAbilities.GetFact(BlueprintRoot.Instance.SystemMechanics.MountAbility);
		if (unitPartRider == null && fact != null && fact.IsOn)
		{
			using (ContextData<ActivatableAbility.RemoveBuffImmediately>.Request().Setup(fact))
			{
				fact.TurnOffImmediately();
			}
		}
		if (unitPartRider != null && unitPartRider.SaddledUnit == null)
		{
			unitPartRider.DismountForce();
			PFLog.Default.ErrorWithReport($"Removed invalid UnitPartRider ({this})");
		}
		UnitPartSaddled unitPartSaddled = Get<UnitPartSaddled>();
		if (unitPartSaddled != null && unitPartSaddled.Rider?.Get<UnitPartRider>()?.SaddledUnit != this)
		{
			unitPartSaddled.ClearAndRemoveSelf();
			PFLog.Default.ErrorWithReport($"Removed invalid UnitPartSaddled ({this})");
		}
	}

	}

	public bool WillTakeTurn()
	{
		if (Descriptor.State.HasCondition(UnitCondition.Dazed))
		{
			return false;
		}
		if (Descriptor.State.HasCondition(UnitCondition.Stunned))
		{
			return false;
		}
		if (Descriptor.State.IsHelpless)
		{
			return false;
		}
		if (Descriptor.State.Prone.Active && UnitProneController.ShouldBeProne(this))
		{
			return false;
		}
		return true;
	}

	public bool HasControlLossEffects()
	{
		if (Descriptor.State.HasCondition(UnitCondition.Confusion))
		{
			return true;
		}
		if (Descriptor.State.HasCondition(UnitCondition.Frightened))
		{
			return true;
		}
		foreach (Buff buff in Buffs)
		{
			foreach (EntityFactComponent component in buff.Components)
			{
				BlueprintComponent sourceBlueprintComponent = component.SourceBlueprintComponent;
				if (sourceBlueprintComponent is ChangeFaction)
				{
					return true;
				}
				if (sourceBlueprintComponent is AddCondition addCondition)
				{
					if (addCondition.Condition == UnitCondition.CantMove)
					{
						return true;
					}
					if (addCondition.Condition == UnitCondition.CanNotAttack)
					{
						return true;
					}
				}
			}
		}
		return false;
	}

	public void SpendAction(UnitCommand.CommandType type, bool isFullRound, float timeSinceCommandStart)
	{
		if (!IsInCombat)
		{
			return;
		}
		UnitCombatState.Cooldowns cooldown = CombatState.Cooldown;
		if (CombatController.IsInTurnBasedCombat())
		{
			switch (type)
			{
			case UnitCommand.CommandType.Move:
				cooldown.MoveAction += 3f;
				break;
			case UnitCommand.CommandType.Standard:
				cooldown.StandardAction += 6f;
				if (isFullRound)
				{
					cooldown.MoveAction += 3f;
				}
				break;
			case UnitCommand.CommandType.Swift:
				cooldown.SwiftAction += 6f;
				break;
			default:
				throw new ArgumentOutOfRangeException();
			case UnitCommand.CommandType.Free:
				break;
			}
		}
		else
		{
			switch (type)
			{
			case UnitCommand.CommandType.Move:
				cooldown.MoveAction = 3f - timeSinceCommandStart;
				break;
			case UnitCommand.CommandType.Standard:
				cooldown.StandardAction = 6f - timeSinceCommandStart;
				cooldown.MoveAction = 3f;
				break;
			case UnitCommand.CommandType.Swift:
				cooldown.SwiftAction = 6f - timeSinceCommandStart;
				break;
			default:
				throw new ArgumentOutOfRangeException();
			case UnitCommand.CommandType.Free:
				break;
			}
		}
	}

	public void SetStealthData(UnitStealth stealthDta)

		Stealth = stealthDta;
	}

	public void MarkNotOptimizableInSave(bool aliveOnly = true)
	{
		if (OptimizationLevel != OptimizationLevel.Never)
		{
			OptimizationLevel = (aliveOnly ? OptimizationLevel.AliveOnly : OptimizationLevel.Never);
		}
	}

	protected override void OnMarkedForDestroy()
	{
		if (Parts.Get<UnitPartCompanion>() != null)
		{
			PFLog.Default.Error($"Trying to destroy {this} who is a companion. Do not do that!");
		}
	}

	public void RecreateView()
	{
		if (!View)
		{
			return;
		}
		UnitEntityView unitEntityView = CreateView();
		if (unitEntityView == null)
		{
			return;
		}
		foreach (Buff buff in Buffs)
		{
			buff.ClearParticleEffect();
		}
		UnitEntityView view = View;
		unitEntityView.UniqueId = base.UniqueId;
		unitEntityView.transform.SetParent(view.transform.parent, worldPositionStays: false);
		unitEntityView.transform.position = view.transform.position;
		unitEntityView.transform.rotation = view.transform.rotation;
		unitEntityView.DisableSizeScaling = true;
		unitEntityView.Blueprint = Blueprint;
		unitEntityView.EnableIK = View.EnableIK;
		AttachToViewOnLoad(unitEntityView);
		Commands.InterruptAll((UnitCommand cmd) => !(cmd is UnitMoveTo));
		Game.Instance.UI.SelectionManager?.ForceCreateMarks();
		UnityEngine.Object.Destroy(view.gameObject);
	}

	public void TempUnmount()
	{
		m_TempUnmountPet = RiderPart.SaddledUnit;
		RiderPart.DismountForce();
	}

	public void RecoveryMount()
	{
		if (m_TempUnmountPet != null)
		{
			Ensure<UnitPartRider>().Mount(m_TempUnmountPet);
			m_TempUnmountPet = null;
		}
	}
}
